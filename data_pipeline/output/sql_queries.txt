======================================================================
Query 1 - SELECT and WHERE
======================================================================
SELECT
    book_id,
    title,
    price_gbp,
    price_inr,
    rating
FROM books
WHERE price_gbp > 40
ORDER BY price_gbp DESC;

======================================================================
Query 2 - ORDER BY and LIMIT
======================================================================
SELECT
    title,
    price_gbp,
    price_inr,
    rating
FROM books
ORDER BY price_gbp DESC
LIMIT 10;

======================================================================
Query 3 - DISTINCT
======================================================================
SELECT DISTINCT
    rating
FROM books
ORDER BY rating;

======================================================================
Query 4 - BETWEEN
======================================================================
SELECT
    title,
    price_gbp,
    price_inr,
    rating
FROM books
WHERE price_gbp BETWEEN 20 AND 40
ORDER BY price_gbp;

======================================================================
Query 5 - IN
======================================================================
SELECT
    title,
    price_gbp,
    price_inr,
    rating
FROM books
WHERE rating IN (4, 5)
ORDER BY
    rating DESC,
    price_gbp DESC;

======================================================================
Query 6 - JOIN
======================================================================
SELECT
    b.book_id,
    b.title,
    b.price_gbp,
    b.price_inr,
    b.rating,
    b.in_stock,
    c.category_id,
    c.category_name
FROM books AS b
INNER JOIN categories AS c
    ON b.category_id = c.category_id
ORDER BY
    c.category_name ASC,
    b.rating DESC,
    b.title ASC;
